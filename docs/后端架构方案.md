## 2. 核心模块设计

### 2.1 VIEW 机制（核心特性）

```
┌─────────────────────────────────────────────────────────────────┐
│                      VIEW 机制架构                                │
└─────────────────────────────────────────────────────────────────┘

┌──────────────┐
│   前端请求   │  SQL
└──────┬───────┘
       │
       ▼
┌─────────────────────────────────────────┐
│  1. 命令层 (commands/query.rs)           │
│     - 接收 SQL                          │
│     - 安全验证                           │
└──────┬─────────────────────────────────┘
       │
       ▼
┌─────────────────────────────────────────┐
│  2. 路由器 (database/router.rs)          │
│     - 判断查询类型                       │
│     - 智能路由到 DuckDB                 │
│     - 无需 SQL 字符串替换（已废弃）      │
└──────┬─────────────────────────────────┘
       │
       ▼
┌─────────────────────────────────────────┐
│  3. DuckDB 执行                        │
│     - 使用预先创建的 VIEW               │
│     - 高效分析查询                       │
│     - 返回结果集                        │
└──────┬─────────────────────────────────┘
       │
       ▼
┌──────────────┐
│   返回前端   │
└──────────────┘
```

**核心设计：**
- **双引擎架构：** SQLite 负责事务型操作（数据写入），DuckDB 负责分析型查询（数据读取）
- **VIEW 同步：** 每次 SQLite 表变更后，自动刷新 DuckDB 中的对应 VIEW
- **零字符串替换：** 彻底移除 SQL 字符串替换，使用原生 VIEW 机制
- **安全隔离：** 用户 SQL 直接在 DuckDB 环境中执行，与 SQLite 事务完全隔离

### 2.3 查询服务 (services/query_service.rs)

```rust
pub struct QueryService;

impl QueryService {
    /// 执行 SQL 查询
    pub fn execute(
        state: &AppState,
        sql: &str,
        use_cache: bool,
    ) -> Result<QueryResult, AppError> {
        // 1. 验证查询（安全检查）
        Self::validate_query(sql)?;
        
        // 2. 检查缓存
        if use_cache {
            if let Some(cached) = CacheService::get(sql) {
                return Ok(cached);
            }
        }
        
        // 3. 路由到 DuckDB（VIEW 机制）
        let result = DatabaseRouter::route_and_execute(state, sql)?;
        
        // 4. 缓存结果
        CacheService::set(sql, &result);
        
        // 5. 记录历史
        HistoryService::record(state, sql, &result);
        
        Ok(result)
    }
    
    /// 验证查询安全性
    fn validate_query(sql: &str) -> Result<(), AppError> {
        // 禁止：DROP、DELETE、ALTER 等危险操作
        // 只允许：SELECT、WITH
        let dangerous = ["DROP", "DELETE", "ALTER", "TRUNCATE", "INSERT", "UPDATE"];
        for keyword in &dangerous {
            if sql.to_uppercase().contains(keyword) {
                return Err(AppError::ForbiddenQuery);
            }
        }
        Ok(())
    }
}
```

### 3.1 连接池管理 (database/pool.rs)

```rust
/// SQLite 连接池（事务型操作）
pub struct SqlitePool {
    pool: r2d2::Pool<SqliteConnectionManager>,
}

/// DuckDB 连接（分析型查询，单连接高性能）
pub struct DuckDbPool {
    conn: Arc<Mutex<duckdb::Connection>>,
    temp_file: NamedTempFile,
}

impl DuckDbPool {
    /// 初始化（创建临时文件 + 绑定 SQLite + 创建 VIEW）
    pub fn init(sqlite_path: &Path) -> Result<Self, AppError> {
        // 1. 创建临时文件
        let temp_file = NamedTempFile::new()?;
        
        // 2. 初始化 DuckDB
        let conn = duckdb::Connection::open(temp_file.path())?;
        
        // 3. ATTACH SQLite
        conn.execute(
            &format!("ATTACH '{}' AS sqlite_db (TYPE SQLITE)", sqlite_path.display()),
            [],
        )?;
        
        // 4. 为每个表创建 VIEW
        let table_names = Self::get_sqlite_tables(&conn)?;
        for table_name in &table_names {
            conn.execute(
                &format!(
                    "CREATE OR REPLACE VIEW {} AS SELECT * FROM sqlite_db.{}",
                    table_name, table_name
                ),
                [],
            )?;
        }
        
        Ok(Self {
            conn: Arc::new(Mutex::new(conn)),
            temp_file,
        })
    }
    
    /// 执行分析查询
    pub fn execute_query(&self, sql: &str) -> Result<Vec<Row>, AppError> {
        let conn = self.conn.lock()?;
        let mut stmt = conn.prepare(sql)?;
        let rows = stmt.query_map([], |row| Self::row_to_map(row))?;
        Ok(rows.collect::<Result<_, _>>()?)
    }
}
```

### 3.2 查询路由器 (database/router.rs)

```rust
pub struct DatabaseRouter;

impl DatabaseRouter {
    /// 路由并执行查询（VIEW 机制）
    pub fn route_and_execute(
        state: &AppState,
        sql: &str,
    ) -> Result<QueryResult, AppError> {
        // 使用 VIEW 机制，所有查询都路由到 DuckDB
        let duckdb = state.duckdb_conn.lock()?;
        if let Some(conn) = &*duckdb {
            let result = Self::execute_duckdb(conn, sql)?;
            Ok(result)
        } else {
            Err(AppError::DatabaseNotInitialized)
        }
    }
    
    /// 在 DuckDB 上执行（使用 VIEW）
    fn execute_duckdb(
        conn: &duckdb::Connection,
        sql: &str,
    ) -> Result<QueryResult, AppError> {
        let start = Instant::now();
        
        let mut stmt = conn.prepare(sql)?;
        let column_names: Vec<String> = stmt
            .column_names()
            .iter()
            .map(|s| s.to_string())
            .collect();
        
        let mut rows = Vec::new();
        let mut row_iter = stmt.query_map([], |row| {
            let mut map = std::collections::HashMap::new();
            for (i, name) in column_names.iter().enumerate() {
                map.insert(name.clone(), row.get(i)?);
            }
            Ok(map)
        })?;
        
        while let Some(row) = row_iter.next() {
            rows.push(row?);
        }
        
        let duration = start.elapsed();
        
        Ok(QueryResult {
            columns: column_names,
            rows,
            duration_ms: duration.as_millis() as u64,
        })
    }
}
```

### 5.1 错误类型定义

```rust
#[derive(thiserror::Error, Debug)]
pub enum AppError {
    #[error("数据库错误: {0}")]

    IoError(#[from] std::io::Error),
    
    #[error("禁止的查询操作")]
    ForbiddenQuery,
    
    #[error("数据库未初始化")]
    DatabaseNotInitialized,
    
    #[error("无效配置: {0}")]
    InvalidConfig(String),
    
    #[error("导入失败: {0}")]
    ImportError(String),
    
    #[error("内部错误: {0}")]
    Internal(String),
}

/// Tauri 错误转换
impl serde::Serialize for AppError {
    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
    where
        S: serde::ser::Serializer,
    {
        serializer.serialize_str(&self.to_string())
    }
}
```

### 5.2 安全机制

| 安全层 | 措施 | 说明 |
|------|------|------|
| **查询验证** | 禁止危险关键字 | 不允许 DROP/DELETE/ALTER 等操作 |
| **路径隔离** | 工作区沙箱 | 所有文件操作限制在工作区目录内 |
| **SQL 注入防护** | VIEW 机制 | 无需字符串替换，原生 SQL 执行 |
| **参数化查询** | 绑定参数 | 防止注入（导入等操作） |
| **输入验证** | 类型检查 | 所有用户输入进行类型和范围验证 |

### 7.2 回收站服务实现

```rust
// src-tauri/src/services/trash_service.rs
use crate::error::AppError;
use crate::models::trash::{TrashConfig, TrashItem, TrashStrategy};
use crate::state::AppState;
use chrono::{Duration, Utc};
use std::path::PathBuf;
use rusqlite::params;

pub struct TrashService;

impl TrashService {
    /// 获取回收站配置
    pub fn get_config(state: &AppState) -> Result<TrashConfig, AppError> {
        let sqlite_conn = state.sqlite_conn.lock()?;
        
        let mut stmt = sqlite_conn.prepare("
            SELECT key, value FROM settings
            WHERE key LIKE 'trash_%'
        ")?;
        
        let mut config = TrashConfig {
            small_table_threshold_rows: 100000,
            default_retention_days: 7,
            auto_clean_enabled: true,
            auto_clean_time: "02:00".to_string(),
        };
        
        let rows = stmt.query_map([], |row| {
            Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
        })?;
        
        for row_result in rows {
            let (key, value) = row_result?;
            match key.as_str() {
                "trash_small_table_threshold_rows" => {
                    config.small_table_threshold_rows = value.parse()?;
                }
                "trash_default_retention_days" => {
                    config.default_retention_days = value.parse()?;
                }
                "trash_auto_clean_enabled" => {
                    config.auto_clean_enabled = value.parse()?;
                }
                "trash_auto_clean_time" => {
                    config.auto_clean_time = value;
                }
                _ => {}
            }
        }
        
        Ok(config)
    }

    /// 确定删除策略
    pub fn determine_strategy(
        row_count: i64,
        config: &TrashConfig,
    ) -> TrashStrategy {
        if row_count <= config.small_table_threshold_rows {
            TrashStrategy::Small
        } else {
            TrashStrategy::Large
        }
    }

    /// 移到回收站
    pub async fn move_to_trash(
        state: &AppState,
        table_name: &str,
    ) -> Result<String, AppError> {
        let config = Self::get_config(state)?;
        let sqlite_conn = state.sqlite_conn.lock().await;
        
        // 1. 获取表信息
        let (table_id, row_count, schema_json): (String, i64, String) = 
            sqlite_conn.query_row(
                "SELECT id, row_count, schema_json FROM tables 
                 WHERE name = ?1 AND status = 'active'",
                params![table_name],
                |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)),
            )?;
        
        // 2. 确定策略
        let strategy = Self::determine_strategy(row_count, &config);
        let trash_expires_at = Utc::now() + Duration::days(config.default_retention_days);
        
        let trash_name = format!("__trash_{}_{}", table_id, table_name);
        let trash_file_path = match strategy {
            TrashStrategy::Small => None,
            TrashStrategy::Large => {
                let workspace = state.workspace_path.lock()?.as_ref().unwrap().clone();
                let trash_dir = workspace.join(".trash");
                std::fs::create_dir_all(&trash_dir)?;
                
                let timestamp = Utc::now().format("%Y%m%d_%H%M%S");
                let file_name = format!("{}_{}.parquet", table_name, timestamp);
                let file_path = trash_dir.join(file_name);
                
                // 导出到 Parquet
                Self::export_table_to_parquet(&sqlite_conn, table_name, &file_path)?;
                
                Some(file_path.to_string_lossy().to_string())
            }
        };
        
        // 3. 执行删除操作
        match strategy {
            TrashStrategy::Small => {
                // 重命名表
                sqlite_conn.execute(
                    &format!("ALTER TABLE {} RENAME TO {}", table_name, trash_name),
                    [],
                )?;
            }
            TrashStrategy::Large => {
                // 直接删除表
                sqlite_conn.execute(&format!("DROP TABLE {}", table_name), [])?;
            }
        }
        
        // 4. 更新元数据
        sqlite_conn.execute(
            "UPDATE tables 
             SET name = ?1,
                 status = 'trashed',
                 original_name = ?2,
                 trashed_at = CURRENT_TIMESTAMP,
                 trash_expires_at = ?3,
                 trash_file_path = ?4,
                 updated_at = CURRENT_TIMESTAMP
             WHERE id = ?5",
            params![
                trash_name,
                table_name,
                trash_expires_at.to_rfc3339(),
                trash_file_path,
                table_id,
            ],
        )?;
        
        // 5. 同步 DuckDB VIEW
        drop(sqlite_conn);
        state.sync_views()?;
        
        Ok(table_id)
    }

    /// 从回收站恢复
    pub async fn restore_from_trash(
        state: &AppState,
        table_id: &str,
    ) -> Result<(), AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        
        // 1. 获取表信息
        let (original_name, trash_name, trash_file_path, schema_json, row_count): 
            (String, String, Option<String>, String, i64) = 
            sqlite_conn.query_row(
                "SELECT original_name, name, trash_file_path, schema_json, row_count 
                 FROM tables 
                 WHERE id = ?1 AND status = 'trashed'",
                params![table_id],
                |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?, row.get(3)?, row.get(4)?)),
            )?;
        
        let config = Self::get_config(state)?;
        let strategy = Self::determine_strategy(row_count, &config);
        
        // 2. 恢复表
        match strategy {
            TrashStrategy::Small => {
                // 重命名回原名称
                sqlite_conn.execute(
                    &format!("ALTER TABLE {} RENAME TO {}", trash_name, original_name),
                    [],
                )?;
            }
            TrashStrategy::Large => {
                // 从 Parquet 导入
                if let Some(file_path) = trash_file_path {
                    Self::import_table_from_parquet(
                        &sqlite_conn,
                        &original_name,
                        PathBuf::from(file_path),
                        &schema_json,
                    )?;
                }
            }
        }
        
        // 3. 更新元数据
        sqlite_conn.execute(
            "UPDATE tables 
             SET name = ?1,
                 status = 'active',
                 original_name = NULL,
                 trashed_at = NULL,
                 trash_expires_at = NULL,
                 trash_file_path = NULL,
                 updated_at = CURRENT_TIMESTAMP
             WHERE id = ?2",
            params![original_name, table_id],
        )?;
        
        // 4. 清理文件（如果是大表策略）
        if strategy == TrashStrategy::Large {
            if let Some(file_path) = trash_file_path {
                let _ = std::fs::remove_file(file_path);
            }
        }
        
        // 5. 同步 DuckDB VIEW
        drop(sqlite_conn);
        state.sync_views()?;
        
        Ok(())
    }

    /// 永久删除
    pub async fn permanently_delete(
        state: &AppState,
        table_id: &str,
    ) -> Result<(), AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        
        // 1. 获取表信息
        let (trash_name, trash_file_path, row_count): 
            (String, Option<String>, i64) = 
            sqlite_conn.query_row(
                "SELECT name, trash_file_path, row_count 
                 FROM tables 
                 WHERE id = ?1 AND status = 'trashed'",
                params![table_id],
                |row| Ok((row.get(0)?, row.get(1)?, row.get(2)?)),
            )?;
        
        let config = Self::get_config(state)?;
        let strategy = Self::determine_strategy(row_count, &config);
        
        // 2. 删除表数据
        match strategy {
            TrashStrategy::Small => {
                sqlite_conn.execute(&format!("DROP TABLE {}", trash_name), [])?;
            }
            TrashStrategy::Large => {
                if let Some(file_path) = trash_file_path {
                    std::fs::remove_file(file_path)?;
                }
            }
        }
        
        // 3. 删除元数据
        sqlite_conn.execute("DELETE FROM tables WHERE id = ?1", params![table_id])?;
        
        Ok(())
    }

    /// 列出回收站内容
    pub async fn list_trash(state: &AppState) -> Result<Vec<TrashItem>, AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        let config = Self::get_config(state)?;
        
        let mut stmt = sqlite_conn.prepare("
            SELECT id, original_name, display_name, row_count, column_count,
                   source_file_type, trashed_at, trash_expires_at, trash_file_path
            FROM tables
            WHERE status = 'trashed'
            ORDER BY trashed_at DESC
        ")?;
        
        let items = stmt.query_map([], |row| {
            let trashed_at: String = row.get(6)?;
            let trash_expires_at: String = row.get(7)?;
            let trashed_at_dt = DateTime::parse_from_rfc3339(&trashed_at)?.with_timezone(&Utc);
            let trash_expires_at_dt = DateTime::parse_from_rfc3339(&trash_expires_at)?.with_timezone(&Utc);
            
            let row_count: i64 = row.get(3)?;
            let strategy = if row_count <= config.small_table_threshold_rows {
                "small".to_string()
            } else {
                "large".to_string()
            };
            
            let now = Utc::now();
            let duration = trash_expires_at_dt.signed_duration_since(now);
            let days_remaining = duration.num_seconds() as f64 / 86400.0;
            
            Ok(TrashItem {
                id: row.get(0)?,
                original_name: row.get(1)?,
                display_name: row.get(2)?,
                row_count,
                column_count: row.get(4)?,
                source_file_type: row.get(5)?,
                trashed_at: trashed_at_dt,
                trash_expires_at: trash_expires_at_dt,
                trash_file_path: row.get(8)?,
                strategy,
                days_remaining,
            })
        })?;
        
        items.collect::<Result<Vec<_>, _>>().map_err(|e| e.into())
    }

    /// 清理过期项
    pub async fn clean_expired(state: &AppState) -> Result<usize, AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        let config = Self::get_config(state)?;
        
        // 1. 获取过期项
        let mut stmt = sqlite_conn.prepare("
            SELECT id, name, trash_file_path, row_count
            FROM tables
            WHERE status = 'trashed'
              AND datetime(trash_expires_at) < datetime('now')
        ")?;
        
        let mut expired_items = Vec::new();
        let rows = stmt.query_map([], |row| {
            Ok((
                row.get::<_, String>(0)?,
                row.get::<_, String>(1)?,
                row.get::<_, Option<String>>(2)?,
                row.get::<_, i64>(3)?,
            ))
        })?;
        
        for row_result in rows {
            expired_items.push(row_result?);
        }
        
        let count = expired_items.len();
        
        // 2. 删除每个过期项
        for (id, trash_name, trash_file_path, row_count) in expired_items {
            let strategy = Self::determine_strategy(row_count, &config);
            
            match strategy {
                TrashStrategy::Small => {
                    sqlite_conn.execute(&format!("DROP TABLE {}", trash_name), [])?;
                }
                TrashStrategy::Large => {
                    if let Some(file_path) = trash_file_path {
                        let _ = std::fs::remove_file(file_path);
                    }
                }
            }
            
            sqlite_conn.execute("DELETE FROM tables WHERE id = ?1", params![id])?;
        }
        
        Ok(count)
    }
```

```rust
// src-tauri/src/services/settings_service.rs
use crate::error::AppError;
use crate::models::settings::*;
use crate::state::AppState;
use rusqlite::params;
use std::collections::HashMap;
use std::fs::File;
use std::io::{Read, Write};
use std::path::PathBuf;

pub struct SettingsService;

impl SettingsService {
    /// 获取所有设置
    pub async fn get_all(state: &AppState) -> Result<AllSettings, AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        
        let mut stmt = sqlite_conn.prepare("SELECT key, value FROM settings")?;
        let rows = stmt.query_map([], |row| {
            Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
        })?;
        
        let mut settings_map = HashMap::new();
        for row_result in rows {
            let (key, value) = row_result?;
            settings_map.insert(key, value);
        }
        
        // 解析各个设置部分
        Ok(Self::map_to_settings(&settings_map))
    }
    
    /// 获取单个或多个设置
    pub async fn get(
        state: &AppState,
        keys: Option<Vec<String>>,
        prefix: Option<String>,
    ) -> Result<HashMap<String, String>, AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        
        let (query, params) = match (keys, prefix) {
            (Some(ks), None) => {
                let placeholders: Vec<String> = ks.iter().map(|_| "?".to_string()).collect();
                (
                    format!(
                        "SELECT key, value FROM settings WHERE key IN ({})",
                        placeholders.join(",")
                    ),
                    ks,
                )
            }
            (None, Some(p)) => (
                "SELECT key, value FROM settings WHERE key LIKE ?".to_string(),
                vec![format!("{}_%", p)],
            ),
            (None, None) => ("SELECT key, value FROM settings".to_string(), vec![]),
            _ => return Err(AppError::InvalidConfig("不能同时指定 keys 和 prefix".into())),
        };
        
        let mut stmt = sqlite_conn.prepare(&query)?;
        let params_ref: Vec<&dyn rusqlite::ToSql> = params.iter().map(|p| p as _).collect();
        
        let rows = stmt.query_map(params_ref.as_slice(), |row| {
            Ok((row.get::<_, String>(0)?, row.get::<_, String>(1)?))
        })?;
        
        let mut result = HashMap::new();
        for row_result in rows {
            let (key, value) = row_result?;
            result.insert(key, value);
        }
        
        Ok(result)
    }
    
    /// 设置单个或多个值
    pub async fn set(
        state: &AppState,
        settings: HashMap<String, String>,
    ) -> Result<Vec<String>, AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        let mut updated_keys = Vec::new();
        
        for (key, value) in settings {
            let count: i64 = sqlite_conn.query_row(
                "SELECT COUNT(*) FROM settings WHERE key = ?",
                [&key],
                |row| row.get(0),
            )?;
            
            if count > 0 {
                sqlite_conn.execute(
                    "UPDATE settings SET value = ?, updated_at = CURRENT_TIMESTAMP WHERE key = ?",
                    params![&value, &key],
                )?;
            } else {
                sqlite_conn.execute(
                    "INSERT INTO settings (key, value) VALUES (?, ?)",
                    params![&key, &value],
                )?;
            }
            
            updated_keys.push(key);
        }
        
        // 更新缓存
        if let Some(cache) = &state.settings_cache {
            let mut cache = cache.lock().await;
            cache.extend(settings);
        }
        
        Ok(updated_keys)
    }
    
    /// 重置设置为默认值
    pub async fn reset(
        state: &AppState,
        keys: Option<Vec<String>>,
        prefix: Option<String>,
    ) -> Result<Vec<String>, AppError> {
        let sqlite_conn = state.sqlite_conn.lock().await;
        
        let default_settings = Self::get_defaults();
        
        let keys_to_reset: Vec<String> = match (keys, prefix) {
            (Some(ks), None) => ks,
            (None, Some(p)) => default_settings
                .keys()
                .filter(|k| k.starts_with(&format!("{}_", p)))
                .cloned()
                .collect(),
            (None, None) => default_settings.keys().cloned().collect(),
            _ => return Err(AppError::InvalidConfig("不能同时指定 keys 和 prefix".into())),
        };
        
        let mut reset_keys = Vec::new();
        for key in keys_to_reset {
            if let Some(default_value) = default_settings.get(&key) {
                let count: i64 = sqlite_conn.query_row(
                    "SELECT COUNT(*) FROM settings WHERE key = ?",
                    [&key],
                    |row| row.get(0),
                )?;
                
                if count > 0 {
                    sqlite_conn.execute(
                        "UPDATE settings SET value = ?, updated_at = CURRENT_TIMESTAMP WHERE key = ?",
                        params![default_value, &key],
                    )?;
                } else {
                    sqlite_conn.execute(
                        "INSERT INTO settings (key, value) VALUES (?, ?)",
                        params![&key, default_value],
                    )?;
                }
                
                reset_keys.push(key);
            }
        }
        
        Ok(reset_keys)
    }

                "SELECT COUNT(*) FROM settings WHERE key = ?",
                [&key],

                    sqlite_conn.execute(

                    sqlite_conn.execute(
                        "INSERT INTO settings (key, value) VALUES (?, ?)",
                        params![&key, &value],
                    )?;
                }
                
                imported_keys.push(key);
            }
        }
        
        Ok(imported_keys)
    }
```

### 9.1 任务调度器架构

```rust
// src-tauri/src/services/task_scheduler.rs
use crate::error::AppError;
use crate::models::automation::*;
use crate::state::AppState;
use cron::Schedule;
use notify::{RecommendedWatcher, Watcher, RecursiveMode};
use std::sync::{Arc, Mutex};
use std::collections::HashMap;
use std::time::{Duration, Instant};
use tokio::sync::mpsc;

/// 任务调度器，管理定时任务和文件监控任务
pub struct TaskScheduler {
    /// 定时任务调度器
    cron_scheduler: Arc<Mutex<CronScheduler>>,
    /// 文件监控器
    file_watchers: Arc<Mutex<HashMap<String, RecommendedWatcher>>>,
    /// 运行中的任务
    running_tasks: Arc<Mutex<HashMap<String, TaskExecution>>>,
    /// 任务事件通道
    event_sender: mpsc::Sender<TaskEvent>,
}

impl TaskScheduler {
    /// 创建新的任务调度器
    pub fn new(event_sender: mpsc::Sender<TaskEvent>) -> Self {
        Self {
            cron_scheduler: Arc::new(Mutex::new(CronScheduler::new())),
            file_watchers: Arc::new(Mutex::new(HashMap::new())),
            running_tasks: Arc::new(Mutex::new(HashMap::new())),
            event_sender,
        }
    }

    /// 启动调度器
    pub async fn start(&self, state: Arc<AppState>) -> Result<(), AppError> {
        // 加载所有启用的任务
        let tasks = Self::load_enabled_tasks(&state).await?;
        
        // 注册任务
        for task in tasks {
            self.register_task(&state, &task).await?;
        }
        
        // 启动调度循环
        self.start_scheduler_loop(state.clone());
        
        Ok(())
    }

    /// 注册单个任务
    pub async fn register_task(&self, state: &AppState, task: &AutomationTask) -> Result<(), AppError> {
        match task.task_type {
            TaskType::Scheduled => {
                self.register_cron_task(task).await?;
            }
            TaskType::FileWatch => {
                self.register_file_watch_task(state, task).await?;
            }
            TaskType::Manual => {
                // 手动任务不需要注册到调度器
            }
        }
        Ok(())
    }

    /// 取消注册任务
    pub async fn unregister_task(&self, task_id: &str) -> Result<(), AppError> {
        // 移除定时任务
        self.cron_scheduler.lock()?.remove_task(task_id);
        
        // 停止文件监控
        if let Some(watcher) = self.file_watchers.lock()?.remove(task_id) {
            drop(watcher);
        }
        
        Ok(())
    }

    /// 执行任务
    pub async fn execute_task(&self, state: &AppState, task_id: &str, trigger_type: TriggerType) -> Result<String, AppError> {
        let task = Self::load_task(state, task_id).await?;
        
        // 创建执行记录
        let execution_id = Self::create_execution_record(state, &task, trigger_type).await?;
        
        // 在后台运行任务
        let state_clone = state.clone();
        let task_clone = task.clone();
        let execution_id_clone = execution_id.clone();
        let event_sender_clone = self.event_sender.clone();
        
        tokio::spawn(async move {
            let _ = Self::run_task_internal(
                state_clone,
                task_clone,
                execution_id_clone,
                event_sender_clone,
            ).await;
        });
        
        Ok(execution_id)
    }

    // 内部方法：实际运行任务
    async fn run_task_internal(
        state: Arc<AppState>,
        task: AutomationTask,
        execution_id: String,
        event_sender: mpsc::Sender<TaskEvent>,
    ) -> Result<(), AppError> {
        let mut executor = TaskExecutor::new(state.clone(), task.clone(), execution_id.clone());
        
        // 发送开始事件
        let _ = event_sender.send(TaskEvent::Started {
            task_id: task.id.clone(),
            execution_id: execution_id.clone(),
        }).await;
        
        // 执行任务步骤
        let result = executor.execute_steps().await;
        
        // 更新执行记录
        let success = result.is_ok();
        let error_msg = result.as_ref().err().map(|e| e.to_string());
        
        Self::update_execution_record(&state, &execution_id, success, error_msg).await?;
        
        // 发送完成事件
        let _ = event_sender.send(TaskEvent::Completed {
            task_id: task.id,
            execution_id,
            success,
        }).await;
        
        // 发送通知
        if let Some(notif_config) = task.notification_config {
            Self::send_notifications(&state, &task, &notif_config, success).await?;
        }
        
        result.map(|_| ())
    }
}
```

```rust
// src/services/metrics_service.rs
use crate::models::metrics::*;
use crate::state::AppState;
use std::sync::{Arc, Mutex};
use std::collections::VecDeque;
use sysinfo::{System, SystemExt};

pub struct MetricsService {
    query_metrics: Arc<Mutex<VecDeque<QueryMetric>>>,
    system: Arc<Mutex<System>>,
}

impl MetricsService {
    pub fn new() -> Self {
        Self {
            query_metrics: Arc::new(Mutex::new(VecDeque::with_capacity(1000))),
            system: Arc::new(Mutex::new(System::new_all())),
        }
    }

    /// 记录查询指标
    pub fn record_query(&self, sql: &str, duration: Duration, row_count: usize, success: bool) {
        let mut metrics = self.query_metrics.lock().unwrap();
        
        // 保留最近 1000 条记录
        if metrics.len() >= 1000 {
            metrics.pop_front();
        }
        
        metrics.push_back(QueryMetric {
            query_id: uuid::Uuid::new_v4().to_string(),
            sql: sql.to_string(),
            duration_ms: duration.as_millis() as u64,
            row_count,
            success,
            timestamp: chrono::Utc::now().to_rfc3339(),
        });
    }

    /// 获取系统整体指标
    pub fn get_system_metrics(&self) -> SystemMetrics {
        let mut system = self.system.lock().unwrap();
        system.refresh_all();
        
        let metrics = self.query_metrics.lock().unwrap();
        
        // 计算查询统计
        let total_queries = metrics.len() as u64;
        let durations: Vec<u64> = metrics.iter().map(|m| m.duration_ms).collect();
        
        let avg_duration = if durations.is_empty() {
            0.0
        } else {
            durations.iter().sum::<u64>() as f64 / durations.len() as f64
        };
        
        // 计算 P95、P99
        let mut sorted_durations = durations.clone();
        sorted_durations.sort();
        
        let p95_index = (sorted_durations.len() as f64 * 0.95) as usize;
        let p99_index = (sorted_durations.len() as f64 * 0.99) as usize;
        
        SystemMetrics {
            cpu_usage_percent: system.global_cpu_info().cpu_usage() as f64,
            memory_used_bytes: system.used_memory(),
            memory_total_bytes: system.total_memory(),
            disk_used_bytes: 0, // 需要额外实现
            disk_total_bytes: 0,
            total_queries,
            queries_per_second: 0.0, // 需要时间窗口计算
            avg_query_duration_ms: avg_duration,
            p95_query_duration_ms: sorted_durations.get(p95_index).copied().unwrap_or(0) as f64,
            p99_query_duration_ms: sorted_durations.get(p99_index).copied().unwrap_or(0) as f64,
            total_tasks: 0,
            running_tasks: 0,
            success_rate: 100.0,
            timestamp: chrono::Utc::now().to_rfc3339(),
        }
    }

    /// 获取慢查询（超过 1 秒）
    pub fn get_slow_queries(&self, threshold_ms: u64) -> Vec<QueryMetric> {
        let metrics = self.query_metrics.lock().unwrap();
        metrics.iter()
            .filter(|m| m.duration_ms > threshold_ms)
            .cloned()
            .collect()
    }
}
```
