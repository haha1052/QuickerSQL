## 第一部分：前端架构设计

### 1.1 前端技术栈

#### 1.1.1 核心技术选型

| 技术 | 版本 | 用途 | 选择理由 |
|------|------|------|---------|
| **React** | 19.x | UI框架 | 生态成熟，团队熟悉 |
| **TypeScript** | 5.x | 类型系统 | 类型安全，提高可维护性 |
| **Vite** | 5.x | 构建工具 | 速度快，配置简单 |
| **TanStack Query** | 5.x | 数据获取 | 缓存、乐观更新、自动重试 |
| **Zustand** | 5.x | 状态管理 | 简单、轻量、TypeScript友好 |
| **Tailwind CSS** | 3.x | 样式框架 | 快速开发，一致性设计 |
| **Radix UI** | - | UI组件库 | 无样式、可访问、灵活 |
| **ECharts** | 5.x | 图表库 | 功能强大，社区活跃 |
| **Monaco Editor** | - | 代码编辑器 | VSCode同款，专业体验 |
| **AG Grid** | - | 数据表格 | 性能优秀，功能全面 |
| **React Hook Form** | - | 表单处理 | 性能好，类型安全 |
| **Zod** | - | 数据验证 | TypeScript原生，类型推断 |

// src/components/sql/SqlEditor.tsx
import { useRef, useCallback } from 'react';
import Editor, { OnMount, OnChange } from '@monaco-editor/react';
import type { editor } from 'monaco-editor';
import { useQuery } from '@/hooks/useQuery';
import { useKeyboard } from '@/hooks/useKeyboard';
import { SqlToolbar } from './SqlToolbar';
import { QueryResult } from './QueryResult';
import { cn } from '@/lib/utils';

export interface SqlEditorProps {
  initialSql?: string;
  onExecute?: (sql: string) => void;
  className?: string;
}

export const SqlEditor = ({
  initialSql = '',
  onExecute,
  className
}: SqlEditorProps) => {
  const editorRef = useRef<editor.IStandaloneCodeEditor | null>(null);
  const { executeQuery, isExecuting, result, error } = useQuery();
  const { registerShortcut } = useKeyboard();

  const handleEditorMount: OnMount = useCallback((editor, monaco) => {
    editorRef.current = editor;
    monaco.languages.registerCompletionItemProvider('sql', {
      provideCompletionItems: (model, position) => {
        return getSqlCompletions(model, position, tableNames);
      }
    });
    editor.addCommand(monaco.KeyMod.CtrlCmd | monaco.KeyCode.Enter, () => {
      handleExecute();
    });
  }, []);

  const handleExecute = useCallback(async () => {
    const sql = editorRef.current?.getValue();
    if (!sql?.trim()) return;
    try {
      await executeQuery(sql);
      onExecute?.(sql);
    } catch (err) {
      // 错误已在 hook 中处理
    }
  }, [executeQuery, onExecute]);

  return (
    <div className={cn('flex flex-col h-full', className)}>
      <SqlToolbar
        onExecute={handleExecute}
        isExecuting={isExecuting}
        onFormat={() => editorRef.current?.getAction('editor.action.formatDocument')?.run()}
        onClear={() => editorRef.current?.setValue('')}
      />
      <div className="flex-1 min-h-0">
        <Editor
          height="100%"
          defaultLanguage="sql"
          value={initialSql}
          onMount={handleEditorMount}
          options={{
            minimap: { enabled: false },
            fontSize: 14,
            lineNumbers: 'on',
            scrollBeyondLastLine: false,
            automaticLayout: true,
            tabSize: 2,
            formatOnPaste: true,
            formatOnType: true,
          }}
          theme="vs-dark"
        />
      </div>
      <QueryResult
        data={result?.data}
        columns={result?.columns}
        rowCount={result?.rowCount}
        executionTime={result?.executionTime}
        error={error}
      />
    </div>
  );
};

### 3.2 数据库连接管理

#### 3.2.1 连接池设计

```rust
// src/database/pool.rs

use std::sync::Arc;
use tokio::sync::RwLock;

/// 数据库连接池
pub struct DatabasePool {
    /// SQLite 连接（主连接）
    sqlite: Arc<Mutex<Box<dyn DatabaseConnection>>>,
    
    /// DuckDB 连接（分析用）
    duckdb: Arc<Mutex<Box<dyn DatabaseConnection>>>,
    
    /// DuckDB 临时文件（保持生命周期）
    duckdb_temp: Arc<Mutex<Option<tempfile::NamedTempFile>>>,
    
    /// 元数据缓存
    metadata_cache: Arc<RwLock<HashMap<String, TableMetadata>>>,
}

impl DatabasePool {
    /// 创建新的连接池
    pub async fn new(sqlite_path: &Path) -> Result<Self, AppError> {
        // 初始化 SQLite
        let sqlite_driver = SqliteDriver::new();
        let sqlite = if sqlite_path.exists() {
            sqlite_driver.connect(sqlite_path)?
        } else {
            sqlite_driver.create_and_connect(sqlite_path)?
        };
        
        // 初始化 DuckDB（带 VIEW 机制）
        let duckdb_temp = tempfile::NamedTempFile::new()?;
        let duckdb_driver = DuckDbDriver::new();
        let duckdb = duckdb_driver.create_database(duckdb_temp.path())?;
        
        // ATTACH SQLite 并创建 VIEW
        Self::setup_duckdb_views(&duckdb, sqlite_path)?;
        
        Ok(Self {
            sqlite: Arc::new(Mutex::new(sqlite)),
            duckdb: Arc::new(Mutex::new(duckdb)),
            duckdb_temp: Arc::new(Mutex::new(Some(duckdb_temp))),
            metadata_cache: Arc::new(RwLock::new(HashMap::new())),
        })
    }
    
    /// 设置 DuckDB VIEW
    fn setup_duckdb_views(
        duckdb: &Box<dyn DatabaseConnection>,
        sqlite_path: &Path
    ) -> Result<(), AppError> {
        // 1. ATTACH SQLite
        let sqlite_str = sqlite_path.to_string_lossy().replace("\\", "\\\\");
        duckdb.execute(&format!(
            "ATTACH '{}' AS sqlite_db (TYPE SQLITE)",
            sqlite_str
        ), &[])?;
        
        // 2. 获取所有表名
        let tables = duckdb.query(
            "SELECT name FROM sqlite_db.sqlite_master WHERE type='table' AND name NOT LIKE 'sqlite_%'",
            &[]
        )?;
        
        // 3. 为每个表创建 VIEW
        for row in tables.rows {
            if let Some(table_name) = row.first() {
                let quoted = format!("\"{}\"", table_name);
                duckdb.execute(&format!(
                    "CREATE OR REPLACE VIEW {} AS SELECT * FROM sqlite_db.{}",
                    quoted, quoted
                ), &[])?;
            }
        }
        
        // 4. 配置 DuckDB
        duckdb.execute("SET preserve_insertion_order = false", &[])?;
        
        Ok(())
    }
    
    /// 获取 SQLite 连接（用于写入）
    pub async fn get_sqlite(&self) -> Result<Arc<Mutex<Box<dyn DatabaseConnection>>>, AppError> {
        Ok(self.sqlite.clone())
    }
    
    /// 获取 DuckDB 连接（用于分析查询）
    pub async fn get_duckdb(&self) -> Result<Arc<Mutex<Box<dyn DatabaseConnection>>>, AppError> {
        Ok(self.duckdb.clone())
    }
    
    /// 重新创建 DuckDB 连接（当 SQLite 数据库变化时）
    pub async fn recreate_duckdb(&self, sqlite_path: &Path) -> Result<(), AppError> {
        let mut duckdb_guard = self.duckdb.lock().map_err(|e| AppError::LockError)?;
        let mut temp_guard = self.duckdb_temp.lock().map_err(|e| AppError::LockError)?;
        
        // 创建新的临时文件
        let new_temp = tempfile::NamedTempFile::new()?;
        let new_path = new_temp.path().to_path_buf();
        
        // 创建新的 DuckDB 连接
        let duckdb_driver = DuckDbDriver::new();
        let new_duckdb = duckdb_driver.create_database(&new_path)?;
        
        // 设置 VIEW
        Self::setup_duckdb_views(&new_duckdb, sqlite_path)?;
        
        // 更新连接
        *duckdb_guard = new_duckdb;
        *temp_guard = Some(new_temp);
        
        Ok(())
    }
    
    /// 获取表元数据（带缓存）
    pub async fn get_table_metadata(&self, table_name: &str) -> Result<TableMetadata, AppError> {
        // 先检查缓存
        {
            let cache = self.metadata_cache.read().await;
            if let Some(meta) = cache.get(table_name) {
                return Ok(meta.clone());
            }
        }
        
        // 缓存未命中，查询数据库
        let sqlite = self.sqlite.lock().await?;
        let meta = Self::fetch_table_metadata(&*sqlite, table_name)?;
        
        // 写入缓存
        {
            let mut cache = self.metadata_cache.write().await;
            cache.insert(table_name.to_string(), meta.clone());
        }
        
        Ok(meta)
    }
    
    /// 使表元数据缓存失效
    pub async fn invalidate_table_metadata(&self, table_name: &str) {
        let mut cache = self.metadata_cache.write().await;
        cache.remove(table_name);
    }
}
```
